# logic-mcp Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** MCP server that lets Claude produce vocal recordings in Logic Pro: session setup, recording, mixing, mastering, bounce, and self-verification by audio analysis.

**Architecture:** Three control layers (MCU emulation over a virtual CoreMIDI port, key commands via synthetic keystrokes, AX scripting via JXA) plus an ffmpeg analysis layer, composed into ~35 MCP tools. Spec: `docs/specs/2026-08-12-logic-mcp-design.md`.

**Tech Stack:** TypeScript strict, @modelcontextprotocol/sdk (stdio), @julusian/midi, osascript JXA, ffmpeg-static, vitest, npm.

## Global Constraints

- Node >= 20, npm, TypeScript strict mode, ESM (`"type": "module"`)
- Logic process name is exactly `Logic Pro X` (version 10.7.9); keep it in ONE constant
- Virtual MIDI port name is exactly `Logic MCP` (both directions)
- All AX selectors and Logic-version-sensitive strings live under `src/ax/` in named constants
- No code comments unless the WHY is non-obvious. No emojis anywhere.
- Stock Logic plugins only for parameter control
- Tools return JSON text content via `ok()`; failures via `fail(message, fixes)` naming the exact user fix
- Every commit message ends with: `Co-Authored-By: Claude Fable 5 <noreply@anthropic.com>`
- Live checkpoints require Anan at the machine with Logic open on a scratch project. Never run live steps against a real song project.
- AX reality rule: Logic's AX tree on macOS 26 may differ from the selectors written here. When a live step fails, run the `ax_dump` helper (Task 8), correct the selector constant, and note the correction in the commit message. Selectors are data, not architecture.

`src/audio/analyze.ts`:
```ts
import { execFile } from 'node:child_process';
import { promisify } from 'node:util';
import { existsSync } from 'node:fs';
import ffmpegPath from 'ffmpeg-static';

const exec = promisify(execFile);

export interface BandEnergy { name: string; lowHz: number; highHz: number; rmsDb: number }
export interface AudioAnalysis {
  lufsIntegrated: number; lufsRange: number; truePeakDb: number;
  rmsDb: number; peakDb: number; durationSec: number; bands: BandEnergy[];
}

export const BANDS = [
  { name: 'lows', lowHz: 20, highHz: 120 },
  { name: 'low-mids', lowHz: 120, highHz: 500 },
  { name: 'mids', lowHz: 500, highHz: 2000 },
  { name: 'presence', lowHz: 2000, highHz: 5000 },
  { name: 'sibilance', lowHz: 5000, highHz: 9000 },
  { name: 'air', lowHz: 9000, highHz: 16000 },
] as const;

function num(m: RegExpMatchArray | null): number {
  if (!m) throw new Error('ffmpeg output missing expected field');
  const v = m[1];
  return v === '-inf' ? -120 : Math.round(parseFloat(v) * 10) / 10;
}

export function parseEbur128(s: string) {
  return {
    lufsIntegrated: num(s.match(/I:\s+(-?[\d.]+|-inf) LUFS/)),
    lufsRange: num(s.match(/LRA:\s+(-?[\d.]+|-inf) LU/)),
    truePeakDb: num(s.match(/Peak:\s+(-?[\d.]+|-inf) dBFS/)),
  };
}

export function parseAstats(s: string) {
  return {
    peakDb: num(s.match(/Peak level dB: (-?[\d.]+|-inf)/)),
    rmsDb: num(s.match(/RMS level dB: (-?[\d.]+|-inf)/)),
  };
}

export function parseDuration(s: string): number {
  const m = s.match(/Duration: (\d+):(\d+):(\d+\.?\d*)/);
  if (!m) throw new Error('ffmpeg output missing duration');
  return parseInt(m[1]) * 3600 + parseInt(m[2]) * 60 + parseFloat(m[3]);
}

async function ffmpegStderr(args: string[]): Promise<string> {
  try {
    const { stderr } = await exec(ffmpegPath as unknown as string, args, { maxBuffer: 32 * 1024 * 1024 });
    return stderr;
  } catch (e: unknown) {
    const err = e as { stderr?: string; message: string };
    if (err.stderr && /Summary|Overall|Duration/.test(err.stderr)) return err.stderr;
    throw new Error(`ffmpeg failed: ${err.message}`);
  }
}

export async function analyzeAudio(filePath: string): Promise<AudioAnalysis> {
  if (!existsSync(filePath)) throw new Error(`File not found: ${filePath}`);
  const main = await ffmpegStderr(['-hide_banner', '-i', filePath, '-af', 'astats=measure_perchannel=none,ebur128=peak=true', '-f', 'null', '-']);
  const bands: BandEnergy[] = [];
  for (const b of BANDS) {
    const s = await ffmpegStderr(['-hide_banner', '-i', filePath, '-af',
      `highpass=f=${b.lowHz}:poles=4,lowpass=f=${b.highHz}:poles=4,astats=measure_perchannel=none`, '-f', 'null', '-']);
    bands.push({ ...b, rmsDb: parseAstats(s).rmsDb });
  }
  return { ...parseEbur128(main), ...parseAstats(main), durationSec: parseDuration(main), bands };
}
```

`src/ax/run-jxa.ts`:

```ts
import { execFile } from 'node:child_process';

export const AX_FIX = 'Grant Accessibility permission: System Settings > Privacy and Security > Accessibility > enable the app hosting this server (your terminal), then retry.';

export class JxaError extends Error {
  constructor(readonly kind: 'ax-permission' | 'no-process' | 'timeout' | 'script', message: string, readonly fixes: string[] = []) {
    super(message);
  }
}

export function runJxa<T = unknown>(body: string, args: unknown = null, timeoutMs = 15000): Promise<T> {
  const src = `ObjC.import('stdlib');
function run(argv) {
  const args = JSON.parse(argv[0]);
  const out = (function(args) { ${body}\n })(args);
  return JSON.stringify(out === undefined ? null : out);
}`;
  return new Promise((resolve, reject) => {
    execFile('osascript', ['-l', 'JavaScript', '-e', src, JSON.stringify(args)],
      { timeout: timeoutMs, maxBuffer: 16 * 1024 * 1024 }, (err, stdout, stderr) => {
        if (err) {
          if (err.killed || /timed out/i.test(err.message)) {
            return reject(new JxaError('timeout', `JXA timed out after ${timeoutMs}ms`));
          }
          const s = String(stderr);
          if (/assistive access|-25211|-1719/.test(s)) {
            return reject(new JxaError('ax-permission', 'Accessibility permission missing.', [AX_FIX]));
          }
          if (/Can't get process|Application isn't running|-600/.test(s)) {
            return reject(new JxaError('no-process', 'Logic Pro X is not running.', ['Open Logic Pro X and load a project.']));
          }
          return reject(new JxaError('script', s.trim() || err.message));
        }
        const text = String(stdout).trim();
        resolve(text ? JSON.parse(text) : (null as T));
      });
  });
}
```

`src/preflight/preflight.ts`:
```ts
import { execFile } from 'node:child_process';
import { promisify } from 'node:util';
import { existsSync } from 'node:fs';
import ffmpegPath from 'ffmpeg-static';
import { runJxa, JxaError, AX_FIX } from '../ax/run-jxa.js';
import { JXA_PRELUDE } from '../ax/prelude.js';
import { LOGIC_PROCESS } from '../constants.js';

const exec = promisify(execFile);

export type Need = 'logic' | 'ax' | 'project' | 'ffmpeg' | 'mcu';

export interface StatusReport {
  logicRunning: boolean; axGranted: boolean; projectOpen: boolean; projectName: string | null;
  ffmpegOk: boolean; mcuConnected: boolean; mcuFeedback: boolean; fixes: string[];
}

let mcuCheck: () => Promise<{ connected: boolean; feedback: boolean }> =
  async () => ({ connected: false, feedback: false });
export function setMcuCheck(fn: typeof mcuCheck) { mcuCheck = fn; }

async function logicRunning(): Promise<boolean> {
  try { await exec('pgrep', ['-x', LOGIC_PROCESS]); return true; } catch { return false; }
}

async function axProbe(): Promise<{ granted: boolean; projectName: string | null }> {
  try {
    const name = await runJxa<string | null>(`${JXA_PRELUDE}
      if (!lp.exists()) return null;
      const wins = lp.windows();
      return wins.length ? lp.windows[0].name() : '';
    `);
    return { granted: true, projectName: name || null };
  } catch (e) {
    if (e instanceof JxaError && e.kind === 'ax-permission') return { granted: false, projectName: null };
    return { granted: true, projectName: null };
  }
}

export async function statusReport(): Promise<StatusReport> {
  const running = await logicRunning();
  const ax = running ? await axProbe() : { granted: false, projectName: null };
  const mcu = await mcuCheck();
  const ffmpegOk = typeof ffmpegPath === 'string' && existsSync(ffmpegPath);
  const fixes: string[] = [];
  if (!running) fixes.push('Open Logic Pro X and load a project.');
  if (running && !ax.granted) fixes.push(AX_FIX);
  if (running && ax.granted && !ax.projectName) fixes.push('Open a Logic project (File > Open).');
  if (!mcu.connected) fixes.push('MIDI port not open yet; call any transport tool once, then re-check.');
  else if (!mcu.feedback) fixes.push('Logic is not talking to the control surface. In Logic: Logic Pro X > Control Surfaces > Setup > New > Install > Mackie Designs Mackie Control > Add, then set both Output Port and Input Port to "Logic MCP".');
  if (!ffmpegOk) fixes.push('ffmpeg-static missing; run npm install in logic-mcp.');
  return {
    logicRunning: running, axGranted: ax.granted, projectOpen: !!ax.projectName,
    projectName: ax.projectName, ffmpegOk, mcuConnected: mcu.connected, mcuFeedback: mcu.feedback, fixes,
  };
}

export async function preflight(needs: Need[]): Promise<void> {
  const r = await statusReport();
  const failWith = (msg: string, fixes: string[]) => {
    const err = new Error(msg) as Error & { fixes: string[] };
    err.fixes = fixes;
    throw err;
  };
  for (const need of needs) {
    if (need === 'logic' && !r.logicRunning) failWith('Logic Pro X is not running.', r.fixes);
    if (need === 'ax' && !r.axGranted) failWith('Accessibility permission missing.', r.fixes);
    if (need === 'project' && !r.projectOpen) failWith('No Logic project open.', r.fixes);
    if (need === 'ffmpeg' && !r.ffmpegOk) failWith('ffmpeg unavailable.', r.fixes);
    if (need === 'mcu' && !r.mcuFeedback) failWith('Control surface link not verified.', r.fixes);
  }
```

```ts
import midi from '@julusian/midi';
import { PORT_NAME } from '../constants.js';

export class McuPorts {
  private input: InstanceType<typeof midi.Input> | null = null;
  private output: InstanceType<typeof midi.Output> | null = null;
  private handlers: ((bytes: number[]) => void)[] = [];

  open(): void {
    if (this.output) return;
    this.input = new midi.Input();
    this.input.ignoreTypes(false, true, true);
    this.input.on('message', (_delta, msg) => {
      for (const h of this.handlers) h([...msg]);
    });
    this.input.openVirtualPort(PORT_NAME);
    this.output = new midi.Output();
    this.output.openVirtualPort(PORT_NAME);
  }
  close(): void {
    this.input?.closePort(); this.output?.closePort();
    this.input = null; this.output = null;
  }
  isOpen(): boolean { return this.output !== null; }
  send(bytes: number[]): void {
    if (!this.output) throw new Error('MIDI port not open');
    this.output.sendMessage(bytes);
  }
  onMessage(cb: (bytes: number[]) => void): void { this.handlers.push(cb); }
}
```

`src/midi/feedback.ts`:
```ts
export class McuState {
  private lcd: string[] = Array(112).fill(' ');
  private tc: string[] = Array(10).fill(' ');
  private leds = new Map<number, boolean>();
  meters: number[] = Array(8).fill(0);
  lastFeedbackAt: number | null = null;

  ingest(bytes: number[]): void {
    if (!bytes.length) return;
    this.lastFeedbackAt = Date.now();
    const status = bytes[0];
    if (status === 0xf0) return this.sysex(bytes);
    const type = status & 0xf0;
    if (type === 0xd0) {
      const ch = bytes[1] >> 4;
      const level = bytes[1] & 0x0f;
      if (ch < 8 && level <= 0x0c) this.meters[ch] = level;
    } else if (type === 0x90) {
      this.leds.set(bytes[1], bytes[2] > 0);
    } else if (type === 0xb0 && bytes[1] >= 0x40 && bytes[1] <= 0x49) {
      const v = bytes[2];
      let c = v & 0x3f;
      if (c < 0x20) c += 0x40;
      this.tc[0x49 - bytes[1]] = String.fromCharCode(c);
    }
  }

  private sysex(b: number[]): void {
    if (b[1] !== 0x00 || b[2] !== 0x00 || b[3] !== 0x66) return;
    if (b[5] !== 0x12) return;
    const pos = b[6];
    for (let i = 7; i < b.length - 1 && pos + i - 7 < 112; i++) {
      this.lcd[pos + i - 7] = String.fromCharCode(b[i]);
    }
  }

  lcdLine(line: 0 | 1): string { return this.lcd.slice(line * 56, line * 56 + 56).join(''); }
  lcdCell(line: 0 | 1, ch: number): string {
    return this.lcd.slice(line * 56 + ch * 7, line * 56 + ch * 7 + 7).join('').trim();
  }
  timecode(): string { return this.tc.join(''); }
  currentBar(): number | null {
    const m = this.timecode().match(/(\d+)/);
    return m ? parseInt(m[1]) : null;
  }
  led(note: number): boolean { return this.leds.get(note) === true; }
}
```

```ts
import { sleep } from '../midi/mcu.js';
```

```ts
export async function insertPlugin(track: string, plugin: string, category?: string): Promise<void> {

  let name = ''; try { name = String(el.name() || el.description() || ''); } catch (e) {}
  if (!name) continue;
  try {
    if (role === 'AXSlider') {
      const p = { name: name, kind: 'slider', value: Number(el.value()) };
      try { p.min = Number(el.attributes['AXMinValue'].value()); p.max = Number(el.attributes['AXMaxValue'].value()); } catch (e) {}
      out.push(p);
    } else if (role === 'AXCheckBox') {
      out.push({ name: name, kind: 'toggle', value: Number(el.value()) === 1 });

    }
  } catch (e) {}

export async function setParams(track: string, slot: number, values: Record<string, number | string | boolean>): Promise<PluginParam[]> {
  await openPluginWindow(track, slot);
  await ensureControlsView();
  await runJxa(`${JXA_PRELUDE}
const win = lp.windows[0];
const wanted = args;
const els = win.entireContents();
const missing = Object.keys(wanted);
for (const el of els) {
  let role; try { role = el.role(); } catch (e) { continue; }
  let name = ''; try { name = String(el.name() || el.description() || ''); } catch (e) {}
  if (!(name in wanted)) continue;
  const v = wanted[name];
  if (role === 'AXSlider') {
    el.value = Number(v);
  } else if (role === 'AXCheckBox') {
    const cur = Number(el.value()) === 1;
    if (cur !== !!v) el.actions['AXPress'].perform();
  } else if (role === 'AXPopUpButton') {
    el.actions['AXPress'].perform();
    delay(0.3);

    must(mi.exists(), 'Menu value not found: ' + v);
    mi.click();
  } else { continue; }
  missing.splice(missing.indexOf(name), 1);
}
must(missing.length === 0, 'Params not found: ' + missing.join(', '));
return true;`, values, 30000);
  return runJxa<PluginParam[]>(PARAMS_SOURCE, null, 30000);
}
```

```ts
import { existsSync, statSync, mkdirSync } from 'node:fs';
import { join } from 'node:path';
import { runJxa } from './run-jxa.js';
import { JXA_PRELUDE } from './prelude.js';
import { clickMenu } from './menus.js';
import { keystroke, keyCode, typeText } from '../keys/keystrokes.js';
import { sleep } from '../midi/mcu.js';

export const BOUNCE_MENU_PATH = ['File', 'Bounce', 'Project or Section…'];

export function bounceDestination(destDir: string, fileName: string) {
  const name = fileName.replace(/\.(wav|aiff|mp3)$/i, '');
  return { dir: destDir, name, full: join(destDir, `${name}.wav`) };
}

async function waitFor(cond: () => Promise<boolean>, timeoutMs: number, what: string): Promise<void> {
  const t0 = Date.now();
  while (Date.now() - t0 < timeoutMs) {
    if (await cond()) return;
    await sleep(400);
  }
  throw new Error(`Timed out waiting for ${what}`);
}

async function bounceDialogReady(): Promise<boolean> {
  return runJxa<boolean>(`${JXA_PRELUDE}
const els = lp.windows[0].entireContents();
for (const el of els) {
  try { if (el.role() === 'AXButton' && (el.name() === 'Bounce' || el.name() === 'OK')) return true; } catch (e) {}
}
return false;`);
}

async function configureDialogPcmWave(): Promise<void> {
  await runJxa(`${JXA_PRELUDE}
const els = lp.windows[0].entireContents();
for (const el of els) {
  try {
    if (el.role() === 'AXCheckBox' && el.name() === 'PCM' && Number(el.value()) !== 1) el.actions['AXPress'].perform();
    if (el.role() === 'AXCheckBox' && (el.name() === 'MP3' || el.name() === 'M4A: AAC') && Number(el.value()) === 1) el.actions['AXPress'].perform();
  } catch (e) {}
}
for (const el of els) {
  try {
    if (el.role() === 'AXPopUpButton' && /AIFF|Wave|CAF/i.test(String(el.value())) && String(el.value()) !== 'Wave') {
      el.actions['AXPress'].perform(); delay(0.3);
      lp.menus[0].menuItems['Wave'].click(); delay(0.3);
    }
  } catch (e) {}
}
return true;`, null, 20000);
}

async function clickDialogButton(name: string): Promise<boolean> {
  return runJxa<boolean>(`${JXA_PRELUDE}
const els = lp.windows[0].entireContents();
for (const el of els) {
  try { if (el.role() === 'AXButton' && el.name() === '${name}') { el.actions['AXPress'].perform(); return true; } } catch (e) {}
}
return false;`);
}

export async function bounce(opts: { destDir: string; fileName: string; timeoutSec?: number }): Promise<string> {
  const dest = bounceDestination(opts.destDir, opts.fileName);
  mkdirSync(dest.dir, { recursive: true });
  await clickMenu(BOUNCE_MENU_PATH);
  await waitFor(bounceDialogReady, 8000, 'bounce dialog');
  await configureDialogPcmWave();
  if (!(await clickDialogButton('Bounce'))) await clickDialogButton('OK');
  await sleep(1200);
  await keystroke('g', ['command', 'shift']);
  await sleep(600);
  await typeText(dest.dir);
  await keyCode(36);
  await sleep(600);
  await runJxa(`${JXA_PRELUDE}
const els = lp.windows[0].entireContents();
for (const el of els) {
  try { if (el.role() === 'AXTextField') { el.value = '${dest.name.replace(/'/g, "\\'")}'; return true; } } catch (e) {}
}
throw new Error('Save panel name field not found');`, null, 15000);
  await sleep(300);
  if (!(await clickDialogButton('Bounce'))) {
    if (!(await clickDialogButton('Save'))) await keyCode(36);
  }
  const timeoutMs = (opts.timeoutSec ?? 300) * 1000;
  await waitFor(async () => {
    if (!existsSync(dest.full)) return false;
    const size1 = statSync(dest.full).size;
    await sleep(1500);
```

```ts
  bankOffset = 0;

  resetBank(): void {
    for (let i = 0; i < 16; i++) this.send(press(NOTE.BANK_LEFT));
    this.bankOffset = 0;
  }

  channelFor(index: number): number {
    const ch = index - this.bankOffset;
    if (ch < 0 || ch > 7) throw new Error(`Track index ${index} outside current bank ${this.bankOffset}-${this.bankOffset + 7}; call ensureVisible first`);
    return ch;
  }

      this.bankOffset += 8;
      await sleep(120);
    }
  }
```

```ts
import { mcu, fader, vpot, press, NOTE, sleep } from './mcu.js';
import { trackIndex } from '../ax/tracks.js';

export function parseDbCell(cell: string): number | null {
  const c = cell.trim();
  if (/^-oo/i.test(c) || c.includes('∞')) return -90;
  const m = c.match(/^([+-]?\d+(?:\.\d+)?)(dB)?$/i);
  return m ? parseFloat(m[1]) : null;
}

export function parsePanCell(cell: string): number | null {
  const c = cell.trim();
  if (c === 'C' || c === '<C>' || c === '0') return 0;
  const m = c.match(/^([LR])\s?(\d+)$/i);
  if (!m) { const n = c.match(/^[+-]?\d+$/); return n ? parseInt(c) : null; }
  return (m[1].toUpperCase() === 'L' ? -1 : 1) * parseInt(m[2]);
}

function readValueCell(ch: number): number | null {
  return parseDbCell(mcu.state.lcdCell(1, ch)) ?? parseDbCell(mcu.state.lcdCell(0, ch));
}

export async function setVolumeDb(trackName: string, db: number, tolDb = 0.3): Promise<number> {
  const i = await trackIndex(trackName);
  await mcu.ensureVisible(i);
  const ch = mcu.channelFor(i);
  mcu.send(press(NOTE.ASSIGN_TRACK));
  await sleep(150);
  let lo = 0, hi = 16383, achieved = NaN;
  for (let iter = 0; iter < 18; iter++) {
    const mid = (lo + hi) >> 1;
    mcu.send(fader(ch, mid));
    await sleep(90);
    const cur = readValueCell(ch);
    if (cur === null) throw new Error('No volume readback on LCD. Verify control surface link with logic_status, then retry.');
    achieved = cur;
    if (Math.abs(cur - db) <= tolDb) break;
    if (cur < db) lo = mid; else hi = mid;
    if (hi - lo <= 1) break;
  }
  return achieved;
}

export async function setPan(trackName: string, pos: number): Promise<number> {
  const i = await trackIndex(trackName);
  await mcu.ensureVisible(i);
  const ch = mcu.channelFor(i);
  mcu.send(press(NOTE.ASSIGN_PAN));
  await sleep(150);
  let cur: number | null = null;
  for (let iter = 0; iter < 140; iter++) {
    cur = parsePanCell(mcu.state.lcdCell(1, ch)) ?? parsePanCell(mcu.state.lcdCell(0, ch));
    if (cur === null) { mcu.send(vpot(ch, 1)); await sleep(60); continue; }
    if (cur === pos) break;
    mcu.send(vpot(ch, cur < pos ? 1 : -1));
    await sleep(50);
  }
  if (cur === null) throw new Error('No pan readback on LCD.');
  return cur;
}

export async function setMute(trackName: string, on: boolean): Promise<boolean> {
  return toggleLed(trackName, NOTE.MUTE_BASE, on);
}
export async function setSolo(trackName: string, on: boolean): Promise<boolean> {
  return toggleLed(trackName, NOTE.SOLO_BASE, on);
}

async function toggleLed(trackName: string, base: number, on: boolean): Promise<boolean> {
  const i = await trackIndex(trackName);
  await mcu.ensureVisible(i);
  const ch = mcu.channelFor(i);
  if (mcu.state.led(base + ch) !== on) { mcu.send(press(base + ch)); await sleep(150); }
  return mcu.state.led(base + ch);
}
```

```ts
export interface Want { pattern: RegExp; value: number | string | boolean }

export function matchParams(live: PluginParam[], wants: Want[]) {
  const applied: Record<string, number | string | boolean> = {};
  const skipped: string[] = [];
  for (const w of wants) {
    const hit = live.find((p) => w.pattern.test(p.name));
    if (hit) applied[hit.name] = w.value;
    else skipped.push(String(w.pattern));
  }
  return { applied, skipped };
}

async function fuzzySet(track: string, slot: number, wants: Want[]) {
  const live = await listParams(track, slot);
  const { applied, skipped } = matchParams(live, wants);
  if (Object.keys(applied).length) await setParams(track, slot, applied);
  return { applied, skipped };
}
```

```ts
export function verdict(a: AudioAnalysis, targetLufs: number, maxTruePeakDb: number) {
  const lufsDelta = Math.round((a.lufsIntegrated - targetLufs) * 10) / 10;
  const peakOk = a.truePeakDb <= maxTruePeakDb;
  let suggestion: string;
  if (Math.abs(lufsDelta) <= 0.5 && peakOk) suggestion = 'On target.';
  else if (lufsDelta < 0) suggestion = `Raise Limiter Gain by about ${Math.abs(lufsDelta)} dB, re-verify.`;
  else suggestion = `Lower Limiter Gain by about ${lufsDelta} dB, re-verify.`;
  if (!peakOk) suggestion += ` True peak ${a.truePeakDb} exceeds ${maxTruePeakDb}; lower the Limiter output ceiling.`;
  return { lufsDelta, peakOk, suggestion };
}
```
